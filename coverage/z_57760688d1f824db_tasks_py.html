14@shared_task 

15def limpar_exif_imagem(evidencia_id): 

16 """ 

17 Recebe o ID de uma evidência recém-salva, abre a imagem em background: 

18 1. Corrige orientação baseada em EXIF (se existir) antes de descartar metadados. 

19 2. Remove 100% dos metadados (EXIF/GPS/Pessoais) gerando nova imagem limpa. 

20 3. Redimensiona proporcionalmente caso a imagem exceda 1920px (LANCZOS). 

21 4. Aplica compressão otimizada (JPEG quality=82 / PNG optimize=True) para economizar disco e banda. 

22 """ 

23 try: 

24 evidencia = Evidencia.objects.get(id=evidencia_id) 

25 caminho_imagem = evidencia.imagem.path 

26 

27 if not os.path.exists(caminho_imagem): 

28 return f"Erro: Arquivo {caminho_imagem} não encontrado no disco." 

29 

30 tamanho_original = os.path.getsize(caminho_imagem) 

31 

32 with Image.open(caminho_imagem) as img: 

33 # Corrige rotação física com base na tag de orientação EXIF antes de removê-la 

34 img = ImageOps.exif_transpose(img) or img 

35 

36 # Identifica o formato pelo arquivo ou pelo cabeçalho 

37 ext = os.path.splitext(caminho_imagem)[1].lower() 

38 if ext in ('.jpg', '.jpeg'): 

39 formato = 'JPEG' 

40 elif ext == '.png': 

41 formato = 'PNG' 

42 elif ext == '.webp': 

43 formato = 'WEBP' 

44 else: 

45 formato = (img.format or 'JPEG').upper() 

46 

47 # Redimensionamento proporcional se ultrapassar a dimensão máxima 

48 if max(img.width, img.height) > MAX_IMAGE_DIMENSION: 

49 resample_filter = getattr(Image, 'Resampling', Image).LANCZOS 

50 img.thumbnail((MAX_IMAGE_DIMENSION, MAX_IMAGE_DIMENSION), resample_filter) 

51 

52 # Garante modo de cor compatível 

53 modo = img.mode 

54 if formato in ('JPEG', 'JPG'): 

55 if modo not in ('RGB', 'L'): 

56 img = img.convert('RGB') 

57 modo = 'RGB' 

58 elif formato == 'PNG': 

59 if modo not in ('RGB', 'RGBA', 'L', 'LA', 'P'): 

60 img = img.convert('RGBA') 

61 modo = 'RGBA' 

62 

63 # Cria nova imagem pura para eliminar metadados e bytes extras 

64 imagem_limpa = Image.new(modo, img.size) 

65 imagem_limpa.paste(img) 

66

68 if formato in ('JPEG', 'JPG'):

71 format='JPEG', 

72 quality=JPEG_QUALITY, 

73 optimize=True, 

74 progressive=True 

75 ) 

76 elif formato == 'PNG': 

77 imagem_limpa.save( 

78 caminho_imagem, 

79 format='PNG', 

80 optimize=True 

81 ) 

82 elif formato == 'WEBP':

87 method=6 

88 ) 

89 else: 

90 imagem_limpa.save(caminho_imagem, format=formato, optimize=True) 

91 

92 tamanho_final = os.path.getsize(caminho_imagem) 

93 economia = ((tamanho_original - tamanho_final) / tamanho_original * 100) if tamanho_original > 0 else 0 

94 

95 msg = (

98 )
